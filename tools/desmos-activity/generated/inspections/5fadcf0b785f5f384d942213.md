## Script 1: Screen 1 g1 (input/graph)

```cl
#narration:"그래프는 x축과 y축의 범위가 -6부터 6까지인 좌표평면을 보여주고 있으며, 좌표평면의 눈금 간격은 1입니다. 좌표평면에는 아홉 개의 동전이 놓여 있습니다."

graph=g1
input=i1
button=b1

#입력한 방정식의 그래프를 그릴 변수를 정의합니다.
m = xyLine(input.latex).slope
b = xyLine(input.latex).yIntercept
x_i = xyLine(input.latex).xIntercept
v = when xyLine(input.latex).isVertical 1 otherwise 0
h = when xyLine(input.latex).isHorizontal 1 otherwise 0

bounds: makeBounds(-5.9, 5.9, -5.9, 5.9)
number("m"): m
number("b"): b
number("x_{in}"): x_i
number("v"): v
number("h"): h

#애니메이션을 제어합니다.
number("t_0"): input.timeSinceSubmit

number("c_{urrentIndex}"): when isUndefined(button.lastValue("c_i")) 0 otherwise button.lastValue("c_i")

#제출한 직선의 방정식을 추적하기 위해 리스트를 생성합니다.
numberList("M_{full}"): when graph.number("c_{urrentIndex}") = input.submitCount graph.numberList("M_i")  otherwise input.history("M_{full}") 
numberList("B_{full}"): when graph.number("c_{urrentIndex}") = input.submitCount graph.numberList("B_i")  otherwise input.history("B_{full}") 
numberList("X_{full}"): when graph.number("c_{urrentIndex}") = input.submitCount graph.numberList("X_i")  otherwise input.history("X_{full}") 
numberList("H_{full}"): when graph.number("c_{urrentIndex}") = input.submitCount graph.numberList("H_i")  otherwise input.history("H_{full}") 
numberList("V_{full}"): when graph.number("c_{urrentIndex}") = input.submitCount graph.numberList("V_i")  otherwise input.history("V_{full}") 

#제출한 직선의 수를 추적합니다.
number("s_{ubmitted}"):when  input.submitted 1 otherwise 0
number("s_{ubmitCount}"): input.submitCount

#고득점을 파악하기 위해 점수를 합산합니다.
numberList("S"): aggregate(graph.number("s"))

readOnly: true


#화면 밖에 있나요?
off = 
  when v = 1 and (x_i<= -5.9 or x_i>=5.9) 1
  when h = 1 and (b<=-5.9 or b>=5.9) 1
  when numericValue("-5.9*${m}+${b}")>=5.9 and numericValue("5.9*${m}+${b}")>=5.9 1
  when numericValue("-5.9*${m}+${b}")<=-5.9 and numericValue("5.9*${m}+${b}")<=-5.9 1
  otherwise 0





trace: false
```

```cl
graph=g1
input=i1


clearOnReset: true
initialLatex:  "y= \\frac{1}{2}x+1"
submitLabel: "수집"

resetLabel: when not(tooMany) and graph.number("h_{itAll}")=0 "다른 직선 추가" when graph.number("h_{itAll}")=1 "" otherwise "${tooManyMsg}"

#이전에 제출한 모든 방정식 데이터를 저장합니다. 리스트에 오류가 발생하지 않도록 정의되지 않은 값은 저장하지 마세요.
capture("M_{full}"): when isUndefined(graph.script.m) 0 otherwise graph.script.m
capture("B_{full}"): when isUndefined(graph.script.b) 0 otherwise graph.script.b
capture("X_{full}"): when isUndefined(graph.script.x_i) 0 otherwise graph.script.x_i
capture("H_{full}"): graph.script.h
capture("V_{full}"): graph.script.v

#경고 및 오류
tooMany = graph.number("c_{urrentLine}")>=10
tooManyMsg="직선이 너무 많습니다! 재시도를 누르고 다시 시도하세요."
notLine = isUndefined(xyLine(input.latex)) and not(isBlank(input.latex))
msg = "x와 y에 대한 직선의 방정식을 입력하세요."


warning: when notLine "${msg}" when tooMany "${tooManyMsg}" otherwise ""
errorMessage: 
  when notLine "${msg}" when tooMany "${tooManyMsg}" 
  when graph.script.off=1 "직선이 그래프 창 밖에 있습니다."
  otherwise ""
  
submitDisabled: tooMany or notLine or graph.number("h_{itAll}")=1



disableEvaluation: true

#모든 동전을 수집했을 때 이 화면이 정확하다고 표시됩니다.
correct: graph.number("h_{itAll}")=1
```
