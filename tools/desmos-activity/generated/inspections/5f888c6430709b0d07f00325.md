## Script 14: Screen 7 note (exhibit/text)

```cl

b = btn
g = graph
in = input1
rotate = simpleFunction(`\operatorname{mod}\left(x+\left(b-y\right),a\right)-b`,`x`,`y`,`a`,`b`)
totalCorrect = when g.script.time > 0 numericValue(`${in.lastValue("correct")}-${in.lastValue("correctC")}`) otherwise in.lastValue("correct")
type         = when b.pressCount = 0 1 when b.pressCount = 1 1 otherwise b.history("type").elementAt(numericValue(`${b.pressCount}-1`)) #Randomize 1 or 2
answer       = when b.pressCount = 0 7 when b.pressCount = 1 -3 otherwise b.history("answer").elementAt(numericValue(`${b.pressCount}-1`)) #Randomize (-10,10)
coefficient  = when b.pressCount = 0 3 when b.pressCount = 1 -6 otherwise rotate.evaluateAt(b.history("coefficient").elementAt(numericValue(`${b.pressCount}-1`)),0,50,25) #Randomize (-25,25)
constant     = when b.pressCount = 0 -13 when b.pressCount = 1 -3 otherwise b.history("constant").elementAt(numericValue(`${b.pressCount}-1`)) #Randomize (-10,10)
type2        = when b.pressCount = 0 1 when b.pressCount = 1 1 when constant = 0 1 otherwise b.history("type2").elementAt(numericValue(`${b.pressCount}-1`)) #when type is 1 type2 must be 2, otherwise it can be random [1,2]
term2Calc = rotate.evaluateAt(b.history("term2").elementAt(numericValue(`${b.pressCount}-1`)),coefficient,20,10)
term2        = when b.pressCount = 0 0 when b.pressCount = 1 0 when totalCorrect <= 3 0 otherwise term2Calc #Take the opposite, then offset by at most 19 (randomize (1,19)
side         = when b.pressCount = 0 2 when b.pressCount = 1 1 otherwise b.history("side").elementAt(numericValue(`${b.pressCount}-1`)) #Randomize 1 or 2
point        = when b.pressCount = 0 2 when b.pressCount = 1 2 otherwise b.history("point").elementAt(numericValue(`${b.pressCount}-1`)) #Randomize 1 or 2
strict       = when b.pressCount = 0 1 when b.pressCount = 1 0 otherwise b.history("strict").elementAt(numericValue(`${b.pressCount}-1`))  #Randomize 0 or 1

type2Value =  when type2 = 1 term2 otherwise numericValue(`${term2}*${answer}`)
loneConstant = 
  when type  = 1 numericValue(`${coefficient}*${answer}+${constant}+${type2Value}`) otherwise numericValue(`${coefficient}*\left(${answer}+${constant}\right)+${type2Value}`)
  
  
side1 = 
  `${when coefficient = 1 `` when coefficient = -1 `-` otherwise `${coefficient}`}${
     when type = 2 and not(constant = 0) `\left(` otherwise ``}x${
     when constant > 0 `+` otherwise ``}${
     when constant = 0 `` otherwise `${constant}`}${
     when type = 2 and not(constant = 0) `\right)` otherwise ``}${
     when term2 > 0 `+` otherwise ``}${
     when (term2 = 1 and type2 = 2) or term2 = 0 `` when (term2 = -1 and type2 = 2) `-` otherwise `${term2}`}${
     when type2 = 2 and not(term2 = 0) `x` otherwise ``}`
     
side2 = `${loneConstant}`

lhs = when side = 1 side1 otherwise side2
rhs = when side = 1 side2 otherwise side1
sybmol = 
  when point = 1 
    when strict = 0 `<` otherwise `\le`
  otherwise
    when strict = 0 `>` otherwise `\ge`

inequality = `${lhs}${sybmol}${rhs}`


boundary = answer
isStrict = strict
direction = when parseInequality(inequality).differenceFunction(`x`).evaluateAt(numericValue(`${answer}+1`)) > 0 1 otherwise -1



shift = when b.pressCount = 0 3 when b.pressCount = 1 -3 otherwise simpleFunction(`x*y`,`x`,`y`).evaluateAt(b.history("shift").elementAt(numericValue(`${b.pressCount}-1`)),b.history("step").elementAt(numericValue(`${b.pressCount}-1`)))
step = when b.pressCount < 2 5 otherwise b.history("step").elementAt(numericValue(`${b.pressCount}-1`))
```
